# Create a new OrganizationAnalyteInterpretationRule
curl -X POST http://localhost:8080/api/interpretation-rules \
-H "Content-Type: application/json" \
-d '{
  "organizationId": 1,
  "analyteId": 1,
  "conditionExpression": "result > 100",
  "classification": "High",
  "autoComment": "Result is high",
  "reflexActionText": "Repeat test",
  "priority": "High"
}'

# Update an existing OrganizationAnalyteInterpretationRule
curl -X PUT http://localhost:8080/api/interpretation-rules/1 \
-H "Content-Type: application/json" \
-d '{
  "classification": "Very High",
  "autoComment": "Result is very high",
  "reflexActionText": "Call doctor",
  "priority": "Critical"
}'

# Delete an existing OrganizationAnalyteInterpretationRule
curl -X DELETE http://localhost:8080/api/interpretation-rules/1
