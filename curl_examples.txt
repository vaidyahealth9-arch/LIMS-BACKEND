# Add or update an analyte for an organization
curl -X POST http://localhost:8080/api/organizations/1/analytes \
-H "Content-Type: application/json" \
-d '{
    "analyteId": 1,
    "price": 15.00,
    "code": "GLU-ORG",
    "bioReference": "60-100 mg/dL"
}'
