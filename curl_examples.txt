# Create a new OrganizationTestInterpretationRule
curl -X POST http://localhost:8080/api/organization-test-interpretation-rules \
-H "Content-Type: application/json" \
-d '{
  "organizationTestId": {
    "organization": {
      "id": 1
    },
    "test": {
      "id": 1
    }
  },
  "conditionExpression": "result > 100",
  "classification": "High",
  "autoComment": "Result is high",
  "reflexActionText": "Repeat test",
  "priority": "High"
}'

# Get a list of OrganizationTestInterpretationRules by organization and test
curl -X GET http://localhost:8080/api/organization-test-interpretation-rules?organizationId=1&testId=1

# Update an existing OrganizationTestInterpretationRule
curl -X PUT http://localhost:8080/api/organization-test-interpretation-rules/1 \
-H "Content-Type: application/json" \
-d '{
  "classification": "Very High",
  "autoComment": "Result is very high",
  "reflexActionText": "Call doctor",
  "priority": "Critical"
}'
